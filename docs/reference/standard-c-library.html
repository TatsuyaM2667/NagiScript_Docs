C 標準ライブラリ

NagiScript から利用可能な C 標準ライブラリを解説します。

現行仕様（v0.2.0）

C 標準関数は extern "C" fn で宣言して利用します。

概要

NagiScript は C ABI と互換のため、C 標準ライブラリの関数を extern "C" fn で宣言してそのまま利用できます。 ただし、以下の点に注意してください：

  • 文字列: C の文字列は null 終端、NagiScript は長さ付き
  • ポインタ: ポインタは *T 記法で宣言し、unsafe ブロック内で操作が必要
  • メモリ管理: C で確保したメモリは C で解放する

stdio.h

基本的な入出力

ngs
extern "C" fn printf(format: *u8) i32
extern "C" fn scanf(format: *u8) i32
extern "C" fn fprintf(stream: *i32, format: *u8) i32
extern "C" fn fscanf(stream: *i32, format: *u8) i32

使用例

ngs
fn main() {
    unsafe {
        printf("Hello, World!\n")
    }
}

stdlib.h

メモリ管理

ngs
extern "C" fn malloc(size: i32) *u8

extern "C" fn realloc(ptr: *u8, size: i32) *u8

使用例

ngs
fn main() {
    unsafe {
        val ptr = malloc(100)

        // メモリを使用
        *ptr = 42

        // メモリを解放
        free(ptr)
    }
}

実践的な例: C ライブラリのラッパー

ngs
// libcurl のラッパー

extern "C" fn curl_easy_init() *u8
extern "C" fn curl_easy_setopt(curl: *u8, option: i32) i32
extern "C" fn curl_easy_perform(curl: *u8) i32
extern "C" fn curl_easy_cleanup(curl: *u8)

struct Curl {
    handle: *u8
}

fn curl_new(handle: *u8) Curl {
    Curl { handle: handle }
}

fn curl_get(curl: Curl) {
    unsafe {
        curl_easy_setopt(curl.handle, 10002)  // CURLOPT_URL
        curl_easy_perform(curl.handle)
    }
}

fn main() {
    unsafe {
        val h = curl_easy_init()
        val c = curl_new(h)
        curl_get(c)
        curl_easy_cleanup(h)
    }
}

注意事項

  1. 文字列の変換: C の文字列は null 終端、NagiScript は長さ付き
  2. メモリ管理: C で確保したメモリは C で解放する
  3. スレッド安全性: C 関数がスレッド安全かどうかを確認
  4. 型の整合性: C と NagiScript の型サイズが一致することを確認
