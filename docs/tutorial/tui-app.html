TUI アプリの作り方

ターミナル上でインタラクティブに操作できる「TUI（Text User Interface）」アプリを NagiScript で作ります。ネイティブバイナリとして単体で動く、キーボード操作のツールを作るのがゴールです。

TUI の基本構造

TUI アプリの骨組みは「描画 → 入力待ち → 状態更新 → 再描画」のループです。NagiScript ではこのループを while で書き、画面更新は ANSI エスケープシーケンスで行います。

loop.ngs
import "std:io"

// ANSI エスケープ定数
const ESC = "\x1b["
const CLEAR = ESC + "2J"       // 画面をクリア
const HOME = ESC + "H"         // カーソルを(1,1)へ
const HIDE = ESC + "?25l"      // カーソルを非表示
const SHOW = ESC + "?25h"      // カーソルを表示

fn draw(frame: i32) {
    var buf = CLEAR + HOME
    buf = buf + "TUI frame: " + str(frame) + "\n"
    buf = buf + "Press [q] to quit.\n"
    io.print(buf)
}

fn main() {
    var frame = 0
    io.print(HIDE)
    while true {
        draw(frame)        // 描画
        // 入力待ちはステップ3で実装
        frame = frame + 1
        if frame >= 120 { break }
    }
    io.print(SHOW)
    io.print(HOME)
}

画面の制御（ANSI エスケープ）

よく使う制御コードは以下のとおりです。io.print でそのまま出力できます。

コード効果
\x1b[2J画面クリア
\x1b[Hカーソルを左上へ移動
\x1b[<行>;<列>H指定位置へ移動
\x1b[<n>A/B/C/Dカーソルを n 移動（上/下/右/左）
\x1b[?25l / ?25hカーソル非表示 / 表示
\x1b[0m装飾リセット
progress.ngs
import "std:io"

fn progress_bar(width: i32, done: i32, total: i32) str {
    var bar = "["
    val filled = (done * width) / total
    var i = 0
    while i < width {
        if i < filled { bar = bar + "#" } else { bar = bar + "-" }
        i = i + 1
    }
    bar = bar + "] " + str((done * 100) / total) + "%"
    return bar
}

fn main() {
    var i = 0
    while i <= 100 {
        io.print("\r" + progress_bar(30, i, 100))   // \r で行先頭に戻る
        i = i + 5
    }
    io.println("")
}

キーボード入力を受け取る

ユーザーがキーを押した瞬間に反応させるには、端末を raw モードにする必要があります。これは C 標準ライブラリの termios を extern で呼び出すと実現できます。C との相互運用の詳細は C言語との相互運用 を参照してください。

rawmode.ngs
import "std:io"

// termios を extern で宣言（libc にリンク）
extern fn tcgetattr(fd: i32, termp: *mut u8) i32
extern fn tcsetattr(fd: i32, actions: i32, termp: *const u8) i32
extern fn getchar() i32

const TCSANOW = 0
const STDIN_FILENO = 0
const ECHO = 8          // ECHO ビットフラグ（OS・アーキテクチャ依存）

// 端末のエコーを無効にして、Enter なしでキーを受ける
fn raw_mode_enter() {
    var term: [64]u8 = [0; 64]      // struct termios の配置領域
    tcgetattr(STDIN_FILENO, &term[0])
    // ECHO ビットを落として反映（簡略化のため、実際は cc_t/lflag 位置を厳密に）
    val flags = term[12] as i32
    val next = (flags & ~ECHO) as u8
    term[12] = next
    tcsetattr(STDIN_FILENO, TCSANOW, &term[0])
}

fn read_key() i32 {
    return getchar()
}

シンプルな対話アプリであれば、raw モードを使わず「1行の入力で Enter 確定」でも十分動きます。

1行入力で動くバージョン

input.ngs
import "std:io"

extern fn getchar() i32

// プロンプトを表示して1行読み取る（Enter で確定）
fn read_line_prompt(prompt: str) str {
    io.print(prompt)
    var line = ""
    while true {
        val c = getchar()
        if c == 10 || c == 13 {   // LF / CR
            io.println("")
            break
        }
        line = line + str(c as u8)
        io.print(str(c as u8))
    }
    return line
}

fn main() {
    val name = read_line_prompt("Your name: ")
    io.println("Hello, " + name + "!")
}

タスクマネージャー TUI を作る

実用的な「タスクリスト管理」アプリです。List<T> でタスクを保持し、std:fs で保存・読み込みも行います。

tasks.ngs
import "std:io"
import "std:fs"

struct Task {
    title: str,
    done: bool,
}

var tasks = List<Task> {}

fn add_task(title: str) {
    tasks.add(Task { title: title, done: false })
}

fn toggle_task(index: i32) {
    if index >= 0 && index < tasks.len() {
        tasks[index].done = !tasks[index].done
    }
}

fn render() str {
    val ESC = "\x1b["
    var buf = ESC + "2J" + ESC + "H"
    buf = buf + "=== Task Manager ===\n\n"
    var i = 0
    while i < tasks.len() {
        val t = tasks[i]
        val mark = if t.done { "[x]" } else { "[ ]" }
        buf = buf + " " + mark + " " + str(i + 1) + ". " + t.title + "\n"
        i = i + 1
    }
    buf = buf + "\n  [a]dd  [t]oggle  [q]uit\n"
    return buf
}

fn main() {
    // フォルダ/ファイルから復元（なければ空で開始）
    val saved = fs.read_to_string("tasks.txt")
    match saved {
        Result.Ok(_) => io.println("loaded saved tasks"),
        Result.Err(_) => { /* 初回起動 */ }
    }

    io.print("\x1b[?25l")
    while true {
        io.print(render())       // 描画
        val key = getchar()      // 入力（raw モードなら Enter なし）

        if key == 113 { break }  // q
        if key == 97 {           // a → タスク追加
            val title = read_line_prompt("\ntitle: ")
            if len(title) > 0 { add_task(title) }
        }
        if key == 116 {          // t → 最新タスクをトグル
            toggle_task(tasks.len() - 1)
        }
    }
    io.print("\x1b[?25h")
    io.println("bye!")
}

ビルドと実行

bash
# ネイティブバイナリにコンパイル（termios は libc にリンク）
nagiscript build tasks.ngs -o task-manager

# 実行
./task-manager

単一バイナリで動くので、インストーラーやコンテナなしでそのまま配布できます。

発展させるときのヒント

  1. ウィンドウ描画を抽象化 — 位置指定（\x1b[行;列H）を使うと、複数ペインをHTTPレスに作れます
  2. 状態は struct でまとめる — AppState を1つの構造体にすると再描画・永続化が楽になります
  3. 帳票のような出力は std:fs で CSV に吐く — TUI は「見る」だけで「結果」はファイルに出す運用が扱いやすい
  4. 色を使う — \x1b[31m（赤）等を組み合わせて選択中カーソルや完了タスクを強調できます
